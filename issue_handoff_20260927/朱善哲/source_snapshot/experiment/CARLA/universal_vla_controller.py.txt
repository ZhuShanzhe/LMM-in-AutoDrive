   52 def _speed_mps(actor: Any) -> float:
   53     velocity = actor.get_velocity()
   54     return math.sqrt(
   55         float(velocity.x) ** 2
   56         + float(velocity.y) ** 2
   57         + float(velocity.z) ** 2
   58     )

   61 def filter_forward_radar_to_route_corridor(
   62     radar_observation: Mapping[str, Any],
   63     *,
   64     ego_x: float,
   65     ego_y: float,
   66     ego_yaw_deg: float,
   67     route_polyline: Sequence[Sequence[float]],
   68     corridor_half_width_m: float,
   69     sensor_offset_m: float = 2.0,
   70 ) -> dict[str, Any]:
   71     """Keep only physical radar returns intersecting the planned corridor.
   72
   73     A narrow forward cone can hit the barrier straight ahead while the route
   74     bends safely away at a junction.  Each binned physical return is projected
   75     into world coordinates and compared with the planner-owned route polyline.
   76     Straight-lane vehicles stay inside the swept corridor; outside barriers
   77     are retained only as audit fields and cannot trigger longitudinal braking.
   78     Missing binned detections fail closed by returning the original aggregate.
   79     """
   80
   81     result = dict(radar_observation)
   82     bins = radar_observation.get("azimuth_obstacle_bins")
   83     try:
   84         points = [
   85             (float(point[0]), float(point[1]))
   86             for point in route_polyline
   87         ]
   88     except (TypeError, ValueError, IndexError):
   89         return result
   90     if not isinstance(bins, list) or len(points) < 2:
   91         return result
   92
   93     yaw_rad = math.radians(float(ego_yaw_deg))
   94     sensor_x = float(ego_x) + float(sensor_offset_m) * math.cos(yaw_rad)
   95     sensor_y = float(ego_y) + float(sensor_offset_m) * math.sin(yaw_rad)
   96     half_width_m = max(0.5, float(corridor_half_width_m))
   97
   98     def distance_to_segment(
   99         point_x: float,
  100         point_y: float,
  101         start: tuple[float, float],
  102         end: tuple[float, float],
  103     ) -> float:
  104         dx = end[0] - start[0]
  105         dy = end[1] - start[1]
  106         norm = dx * dx + dy * dy
  107         if norm <= 1e-9:
  108             return math.hypot(point_x - start[0], point_y - start[1])
  109         fraction = (
  110             (point_x - start[0]) * dx + (point_y - start[1]) * dy
  111         ) / norm
  112         fraction = max(0.0, min(1.0, fraction))
  113         return math.hypot(
  114             point_x - (start[0] + fraction * dx),
  115             point_y - (start[1] + fraction * dy),
  116         )
  117
  118     relevant: list[dict[str, Any]] = []
  119     for raw in bins:
  120         if not isinstance(raw, Mapping):
  121             continue
  122         try:
  123             distance_m = float(raw["distance_m"])
  124             azimuth_rad = math.radians(float(raw["azimuth_deg"]))
  125         except (KeyError, TypeError, ValueError):
  126             continue
  127         if not math.isfinite(distance_m) or distance_m <= 0.0:
  128             continue
  129         point_x = sensor_x + distance_m * math.cos(yaw_rad + azimuth_rad)
  130         point_y = sensor_y + distance_m * math.sin(yaw_rad + azimuth_rad)
  131         lateral_distance_m = min(
  132             distance_to_segment(point_x, point_y, start, end)
  133             for start, end in zip(points, points[1:])
  134         )
  135         if lateral_distance_m <= half_width_m:
  136             item = dict(raw)
  137             item["route_lateral_distance_m"] = round(
  138                 lateral_distance_m, 3
  139             )
  140             relevant.append(item)
  141
  142     nearest = min(
  143         relevant,
  144         key=lambda item: float(item["distance_m"]),
  145         default=None,
  146     )
  147     closing = [
  148         item
  149         for item in relevant
  150         if float(item.get("closing_speed_mps", 0.0) or 0.0) > 0.5
  151     ]
  152     nearest_closing = min(
  153         closing,
  154         key=lambda item: float(item["distance_m"]),
  155         default=None,
  156     )
  157     result.update(
  158         route_corridor_filter_applied=True,
  159         route_corridor_half_width_m=round(half_width_m, 3),
  160         route_corridor_obstacle_bins=relevant,
  161         unfiltered_obstacle_candidate_count=radar_observation.get(
  162             "obstacle_candidate_count"
  163         ),
  164         unfiltered_nearest_distance_m=radar_observation.get(
  165             "nearest_distance_m"
  166         ),
  167         unfiltered_nearest_azimuth_deg=radar_observation.get(
  168             "nearest_azimuth_deg"
  169         ),
  170         obstacle_candidate_count=len(relevant),
  171         closing_candidate_count=len(closing),
  172         nearest_distance_m=(
  173             round(float(nearest["distance_m"]), 3) if nearest else None
  174         ),
  175         nearest_relative_velocity_mps=(
  176             round(float(nearest.get("relative_velocity_mps", 0.0)), 3)
  177             if nearest else None
  178         ),
  179         nearest_azimuth_deg=(
  180             round(float(nearest["azimuth_deg"]), 3) if nearest else None
  181         ),
  182         nearest_closing_distance_m=(
  183             round(float(nearest_closing["distance_m"]), 3)
  184             if nearest_closing else None
  185         ),
  186         nearest_closing_velocity_mps=(
  187             round(float(nearest_closing["closing_speed_mps"]), 3)
  188             if nearest_closing else None
  189         ),
  190     )
  191     return result

  194 def fuse_forward_radar_risk(
  195     learned_risk: Mapping[str, Any],
  196     radar_observation: Mapping[str, Any],
  197     *,
  198     ego_speed_kmh: float,
  199 ) -> dict[str, Any]:
  200     """Fuse a physical forward-radar safety envelope with learned risk.
  201
  202     This is a scene-agnostic safety cage.  It uses only the ego speed and a
  203     narrow forward physical-radar return; no actor truth, scenario ids, event
  204     ids or scheduled distances are accepted by the function.
  205     """
  206
  207     result = dict(learned_risk)
  208     speed_mps = max(0.0, float(ego_speed_kmh) / 3.6)
  209     # Emergency envelope: standstill buffer + controller/sensor reaction
  210     # travel + braking distance at 6 m/s^2.  The six-metre floor keeps a
  211     # stopped vehicle latched instead of allowing the liveness gate to crawl
  212     # back into a close obstacle.
  213     emergency_distance_m = max(
  214         6.0,
  215         1.8 + speed_mps * 0.65 + speed_mps * speed_mps / (2.0 * 6.0),
  216     )
  217     # Earlier comfortable-deceleration envelope at 3.5 m/s^2.
  218     caution_distance_m = max(
  219         12.0,
  220         2.5 + speed_mps + speed_mps * speed_mps / (2.0 * 3.5),
  221     )
  222     result["physical_forward_radar"] = {
  223         **dict(radar_observation),
  224         "emergency_distance_m": round(emergency_distance_m, 3),
  225         "caution_distance_m": round(caution_distance_m, 3),
  226     }
  227     learned_probabilities = dict(result.get("probabilities") or {})
  228     if learned_probabilities:
  229         result["learned_probabilities"] = learned_probabilities
  230     reasons = list(result.get("reason_codes") or [])
  231     # Preserve an explicit no-obstacle snapshot for the temporal supervisor.
  232     # Without the thresholds and sensor frame, a learned visual false positive
  233     # cannot be cleared even after multiple physical empty-road frames.
  234     distance_value = radar_observation.get("nearest_distance_m")
  235     if not isinstance(distance_value, (int, float)):
  236         return result
  237     distance_m = float(distance_value)
  238     if not math.isfinite(distance_m) or distance_m <= 0.0:
  239         return result
  240
  241     if distance_m <= emergency_distance_m:
  242         reasons.append("physical_forward_radar_emergency_distance")
  243         result.update(
  244             risk_level="high",
  245             raw_argmax_level=str(result.get("raw_argmax_level", "low")),
  246             recommended_action="emergency_brake",
  247             reason_codes=list(dict.fromkeys(reasons)),
  248             source="learned_visual_risk+physical_forward_radar",
  249             probabilities={"low": 0.0, "medium": 0.0, "high": 1.0},
  250             risk_score=1.0,
  251             horizon_probabilities=[1.0, 1.0, 0.0],
  252         )
  253         return result
  254
  255     learned_level = str(result.get("risk_level", "low")).lower()
  256     if distance_m <= caution_distance_m and learned_level == "low":
  257         reasons.append("physical_forward_radar_caution_distance")
  258         result.update(
  259             risk_level="medium",
  260             raw_argmax_level=str(result.get("raw_argmax_level", "low")),
  261             recommended_action="decelerate",
  262             reason_codes=list(dict.fromkeys(reasons)),
  263             source="learned_visual_risk+physical_forward_radar",
  264             probabilities={"low": 0.0, "medium": 1.0, "high": 0.0},
  265             risk_score=max(float(result.get("risk_score", 0.0) or 0.0), 0.5),
  266         )
  267     return result

  270 def assess_rear_radar_collision(
  271     radar_observation: Mapping[str, Any],
  272 ) -> dict[str, Any]:
  273     """Classify a rapidly closing rear radar return using physical TTC."""
  274
  275     distance_value = radar_observation.get("nearest_closing_distance_m")
  276     closing_value = radar_observation.get("nearest_closing_velocity_mps")
  277     if not isinstance(distance_value, (int, float)):
  278         nearest_velocity = radar_observation.get(
  279             "nearest_relative_velocity_mps"
  280         )
  281         if isinstance(nearest_velocity, (int, float)) and float(
  282             nearest_velocity
  283         ) < -0.5:
  284             distance_value = radar_observation.get("nearest_distance_m")
  285             closing_value = -float(nearest_velocity)
  286     assessment = {
  287         "schema_version": "rear_collision_assessment/1.0",
  288         "collision_risk": False,
  289         "distance_m": None,
  290         "closing_speed_mps": None,
  291         "ttc_s": None,
  292         "reason": "no_closing_rear_return",
  293     }
  294     if not all(
  295         isinstance(value, (int, float))
  296         for value in (distance_value, closing_value)
  297     ):
  298         return assessment
  299     distance_m = float(distance_value)
  300     closing_speed_mps = float(closing_value)
  301     if (
  302         not math.isfinite(distance_m)
  303         or not math.isfinite(closing_speed_mps)
  304         or distance_m <= 0.0
  305         or closing_speed_mps <= 0.0
  306     ):
  307         return assessment
  308     ttc_s = distance_m / closing_speed_mps
  309     collision_risk = (
  310         closing_speed_mps >= 2.0
  311         and distance_m <= 30.0
  312         and ttc_s <= 3.0
  313     ) or (
  314         closing_speed_mps >= 1.0
  315         and distance_m <= 6.0
  316         and ttc_s <= 4.0
  317     )
  318     assessment.update(
  319         collision_risk=collision_risk,
  320         distance_m=round(distance_m, 3),
  321         closing_speed_mps=round(closing_speed_mps, 3),
  322         ttc_s=round(ttc_s, 3),
  323         reason=(
  324             "rear_radar_ttc_collision_risk"
  325             if collision_risk
  326             else "rear_radar_outside_collision_envelope"
  327         ),
  328     )
  329     return assessment

  332 def apply_directional_collision_response(
  333     final_decision: Mapping[str, Any],
  334     *,
  335     front_risk: Mapping[str, Any] | None,
  336     forward_radar: Mapping[str, Any],
  337     rear_radar: Mapping[str, Any],
  338     ego_speed_kmh: float,
  339     road_speed_limit_kmh: float,
  340     route_speed_cap_kmh: float,
  341     lane_options: Mapping[str, Mapping[str, Any]] | None = None,
  342     allow_evasive_motion: bool = True,
  343 ) -> tuple[dict[str, Any], dict[str, Any], str | None]:
  344     """Apply front-first, speed-limited rear-collision avoidance.
  345
  346     Front risk always wins.  A rear TTC threat may raise speed only below
  347     both the map speed limit and the route controller cap.  At that cap it
  348     may select an adjacent lane only when static-map legality and a learned
  349     side-view risk probe both pass.
  350     """
  351
  352     decision = dict(final_decision)
  353     rear = assess_rear_radar_collision(rear_radar)
  354     assessment: dict[str, Any] = {
  355         "schema_version": "directional_collision_response/1.0",
  356         "front_priority": True,
  357         "rear": rear,
  358         "front_risk_level": str(
  359             (front_risk or {}).get("risk_level", "unknown")
  360         ).lower(),
  361         "selected_response": "none",
  362     }
  363     if not rear["collision_risk"]:
  364         return decision, assessment, None
  365
  366     front_level = assessment["front_risk_level"]
  367     front_distance = forward_radar.get("nearest_distance_m")
  368     front_caution = forward_radar.get("caution_distance_m")
  369     physical_front_conflict = all(
  370         isinstance(value, (int, float))
  371         for value in (front_distance, front_caution)
  372     ) and float(front_distance) <= float(front_caution)
  373     if front_level in {"medium", "high"} or physical_front_conflict:
  374         assessment["selected_response"] = "front_brake_priority"
  375         return decision, assessment, None
  376     if not allow_evasive_motion:
  377         assessment["selected_response"] = "text_hazard_stop_priority"
  378         return decision, assessment, None
  379
  380     caps = [
  381         float(value)
  382         for value in (road_speed_limit_kmh, route_speed_cap_kmh)
  383         if isinstance(value, (int, float))
  384         and math.isfinite(float(value))
  385         and float(value) > 0.0
  386     ]
  387     if not caps:
  388         assessment["selected_response"] = "missing_speed_limit"
  389         return decision, assessment, None
  390     speed_cap_kmh = min(caps)
  391     assessment["speed_cap_kmh"] = round(speed_cap_kmh, 3)
  392     if float(ego_speed_kmh) < speed_cap_kmh - 1.5:
  393         target_speed_kmh = min(
  394             speed_cap_kmh,
  395             max(
  396                 float(ego_speed_kmh) + 8.0,
  397                 float(decision.get("target_speed_kmh", 0.0) or 0.0),
  398             ),
  399         )
  400         decision.update(
  401             action="accelerate",
  402             target_speed_kmh=target_speed_kmh,
  403             target_lane=None,
  404             emergency=False,
  405             reason="physical_rear_radar_acceleration_escape",
  406             blocked_reason_codes=[],
  407         )
  408         assessment["selected_response"] = "accelerate_within_speed_limit"
  409         return decision, assessment, "physical_rear_radar_acceleration_escape"
  410
  411     safe_lanes: list[tuple[float, str]] = []
  412     for direction in ("left", "right"):
  413         option = dict((lane_options or {}).get(direction) or {})
  414         side_risk = dict(option.get("risk") or {})
  415         if not bool(option.get("legal", False)):
  416             continue
  417         if str(side_risk.get("risk_level", "high")).lower() != "low":
  418             continue
  419         high_probability = float(
  420             (side_risk.get("probabilities") or {}).get("high", 0.0) or 0.0
  421         )
  422         safe_lanes.append((high_probability, direction))
  423     if safe_lanes:
  424         _probability, direction = min(safe_lanes)
  425         override = f"physical_rear_radar_lane_change_{direction}_escape"
  426         decision.update(
  427             action=f"lane_change_{direction}",
  428             target_speed_kmh=speed_cap_kmh,
  429             target_lane=direction,
  430             emergency=False,
  431             reason=override,
  432             blocked_reason_codes=[],
  433         )
  434         assessment["selected_response"] = f"safe_{direction}_lane_change"
  435         return decision, assessment, override
  436
  437     assessment["selected_response"] = "hold_no_legal_clear_escape"
  438     return decision, assessment, None

  441 def build_sensor_policy_state(
  442     world: Any,
  443     ego: Any,
  444     *,
  445     frame_id: str,
  446 ) -> dict[str, Any]:
  447     """Build the policy state without enumerating CARLA actors or objects."""
  448
  449     ego_acceleration = ego.get_acceleration()
  450     ego_angular = ego.get_angular_velocity()
  451     control = ego.get_control()
  452     snapshot = world.get_snapshot()
  453     map_waypoint = world.get_map().get_waypoint(
  454         ego.get_location(),
  455         project_to_road=True,
  456         lane_type=__import__("carla").LaneType.Driving,
  457     )
  458     return {
  459         "schema_version": "unified_sensor_policy_state/1.0",
  460         "frame_id": frame_id,
  461         "timestamp_s": float(snapshot.timestamp.elapsed_seconds),
  462         "ego": {
  463             "speed_mps": _speed_mps(ego),
  464             "acceleration_mps2": math.sqrt(
  465                 float(ego_acceleration.x) ** 2
  466                 + float(ego_acceleration.y) ** 2
  467                 + float(ego_acceleration.z) ** 2
  468             ),
  469             "yaw_rate_rps": math.radians(float(ego_angular.z)),
  470             "speed_limit_mps": float(ego.get_speed_limit()) / 3.6,
  471             "control": {
  472                 "steer": float(control.steer),
  473                 "throttle": float(control.throttle),
  474                 "brake": float(control.brake),
  475             },
  476             "is_junction": bool(
  477                 map_waypoint is not None
  478                 and bool(getattr(map_waypoint, "is_junction", False))
  479             ),
  480         },
  481         "environment": {"at_junction": False},
  482         "objects": [],
  483     }

  486 def vehicle_state_tensor(
  487     world: Any,
  488     ego: Any,
  489     *,
  490     include_junction: bool = True,
  491 ) -> torch.Tensor:
  492     """Return the fixed 8-dim vehicle-state modality."""
  493
  494     policy_state = build_sensor_policy_state(
  495         world,
  496         ego,
  497         frame_id="vehicle_state",
  498     )
  499     ego_state = policy_state["ego"]
  500     controls = ego_state["control"]
  501     values = [
  502         float(ego_state["speed_mps"]),
  503         float(ego_state["acceleration_mps2"]),
  504         float(ego_state["yaw_rate_rps"]),
  505         float(controls.get("steer", 0.0)),
  506         float(controls.get("throttle", 0.0)),
  507         float(controls.get("brake", 0.0)),
  508         float(ego_state["speed_limit_mps"]),
  509         float(bool(ego_state.get("is_junction", False)))
  510         if include_junction
  511         else 0.0,
  512     ]
  513     return torch.tensor([values], dtype=torch.float32)

  516 def resolve_sensor_tick(config, physics_dt_s, decision_interval_frames):
  517     decision_dt = float(physics_dt_s) * int(decision_interval_frames)
  518     tick = float(config.get('sensor_tick_s',decision_dt))
  519     if not math.isfinite(tick) or not float(physics_dt_s) <= tick <= decision_dt:
  520         raise ValueError('Sensor tick must lie between physics and decision periods')
  521     if abs(tick/float(physics_dt_s)-round(tick/float(physics_dt_s))) > 1e-6:
  522         raise ValueError('Sensor tick must be an integer number of physics steps')
  523     return tick

  526 def environment_feature_tensor(
  527     world: Any,
  528     ego: Any | None = None,
  529     control_speed_cap_kmh: float | None = None,
  530 ) -> torch.Tensor:
  531     """Return the fixed 14-dim environment-state modality."""
  532
  533     weather = world.get_weather()
  534     road_speed_limit_kmh = (
  535         float(ego.get_speed_limit()) if ego is not None else 100.0
  536     )
  537     control_cap = (
  538         float(control_speed_cap_kmh)
  539         if control_speed_cap_kmh is not None
  540         else road_speed_limit_kmh
  541     )
  542     values = (
  543         float(weather.cloudiness) / 100.0,
  544         float(weather.precipitation) / 100.0,
  545         float(weather.precipitation_deposits) / 100.0,
  546         float(weather.wind_intensity) / 100.0,
  547         float(weather.sun_azimuth_angle) / 360.0,
  548         max(-1.0, min(1.0, float(weather.sun_altitude_angle) / 90.0)),
  549         float(weather.fog_density) / 100.0,
  550         float(weather.fog_distance) / 1000.0,
  551         0.02,
  552         float(weather.wetness) / 100.0,
  553         0.10,
  554         0.80,
  555         min(1.0, max(0.0, road_speed_limit_kmh / 100.0)),
  556         min(1.0, max(0.0, control_cap / 100.0)),
  557     )
  558     return torch.tensor([values], dtype=torch.float32)

  561 def is_initial_sensor_warmup_error(error: Exception, decision_count: int) -> bool:
  562     """Identify the expected safe hold before the first RGB bundle arrives."""
  563
  564     return (
  565         int(decision_count) == 0
  566         and isinstance(error, RuntimeError)
  567         and str(error) == "no synchronized multiview RGB frame is available"
  568     )

  575         self,
  576         *,
  577         world: Any,
  578         ego: Any,
  579         route_controller: Any,

  834     def _lane_command_observation(self,command,parsed):
  835         self._lane_target_entered=False
  836         direction=parsed.requested_lane_direction
  837         if direction not in ('left','right'):
  838             self._lane_command_key=None
  839             return parsed,None
  840         from control.pid_controller import EgoPIDController
  841         key=(str(command.get('id','')),parsed.source_text)
  842         location=self.ego.get_location()
  843         waypoint=self.world.get_map().get_waypoint(location)
  844         if key!=self._lane_command_key:
  845             self._lane_command_key=key
  846             target=EgoPIDController._adjacent_driving_lane(waypoint,'lane_change_'+direction)
  847             self._lane_goal_waypoint=target
  848             self._lane_goal=None if target is None else (target.road_id,target.section_id,target.lane_id)
  849             self._lane_goal_stable=0
  850             self._lane_goal_complete=False
  851             self._lane_change_issued=False
  852         if self._lane_goal is None:return parsed,[]
  853         if (waypoint.road_id,waypoint.section_id)!=self._lane_goal[:2]:
  854             # Follow the latched lane's actual successors across road sections.
  855             from lightweight_vla_adapter.src.lane_risk_contract import continued_lane_waypoint
  856             target=continued_lane_waypoint(self._lane_goal_waypoint,waypoint,location)
  857             if target is not None:
  858                 self._lane_goal=(target.road_id,target.section_id,target.lane_id)
  859                 self._lane_goal_waypoint=target
  860         yaw=math.radians(waypoint.transform.rotation.yaw)
  861         offset=location-waypoint.transform.location
  862         centered=abs(-math.sin(yaw)*offset.x+math.cos(yaw)*offset.y)<.5
  863         self._lane_target_entered=(waypoint.road_id,waypoint.section_id,waypoint.lane_id)==self._lane_goal
  864         arrived=self._lane_target_entered and centered
  865         self._lane_goal_stable=self._lane_goal_stable+1 if arrived else 0
  866         self._lane_goal_complete=self._lane_goal_complete or self._lane_goal_stable>=5
  867         if self._lane_goal_complete:
  868             return replace(parsed,parsed_intent='KEEP_LANE',requested_lane_direction=None),None
  869         target=(waypoint if self._lane_target_entered else EgoPIDController._lane_with_id(waypoint,self._lane_goal[2])) if (waypoint.road_id,waypoint.section_id)==self._lane_goal[:2] else None
  870         if target is None:return parsed,[]
  871         self._lane_goal_waypoint=target
  872         points=[]
  873         for distance in (-10.,0.,10.,20.,30.):
  874             choices=target.previous(-distance) if distance<0 else target.next(distance) if distance>0 else [target]
  875             if choices:points.append(choices[0].transform.location)
  876         tf=self.ego.get_transform();angle=math.radians(tf.rotation.yaw)
  877         return parsed,[[math.cos(angle)*(p.x-location.x)+math.sin(angle)*(p.y-location.y),
  878             -math.sin(angle)*(p.x-location.x)+math.cos(angle)*(p.y-location.y)] for p in points]

  880     def predict_target_lane_risk(
  881         self,
  882         batch: Any,
  883         direction: str,
  884     ) -> dict[str, Any]:
  885         """Probe the learned risk head on one camera view only.
  886
  887         ``direction`` must be ``"left"`` or ``"right"``.  The probe reuses the
  888         same checkpoint and risk head, selects the view purely through
  889         ``camera_view_mask``, and never overwrites the primary fused-view
  890         temporal risk state.
  891         """
  892
  893         if direction not in {"front", "left", "right"}:
  894             raise ValueError("direction must be 'front', 'left' or 'right'")
  895         cache=getattr(self,'_direction_risk_cache',{})
  896         if direction in cache:
  897             return dict(cache[direction])
  898         view_name = direction
  899         if view_name not in self.available_cameras:
  900             return {
  901                 "risk_level": "low",
  902                 "recommended_action": "keep_lane",
  903                 "reason_codes": [],
  904                 "matched_entity_id": None,
  905                 "source": "unavailable_view",
  906             }
  907         view_index = CAMERA_ORDER.index(view_name)
  908         masked_view_mask = torch.zeros_like(batch.camera_view_mask)
  909         masked_view_mask[:, view_index] = batch.camera_view_mask[
  910             :, view_index
  911         ]
  912         masked_batch = replace(
  913             batch,
  914             camera_view_mask=masked_view_mask,
  915         )
  916         risk = self.pipeline.predict_visual_risk(masked_batch)
  917         risk["source"] = f"learned_{view_name}_camera_visual_risk_head"
  918         if hasattr(self,'_direction_risk_cache'):
  919             self._direction_risk_cache[direction]=dict(risk)
  920         return risk

  932     def _decide(self, frame: int) -> None:
  933         decision_call_started=time.perf_counter()

  935         progress_m = self._progress_m()
  936         command = self.fsm.active_command(self.commands, progress_m)

  945                                     self._execution_wait_feedback, self.driving_plan.document['request_id'],
  946                                     self.driving_plan.timestamp_s))

  948                 float(self.world.get_snapshot().timestamp.elapsed_seconds), feedback)
  949         parsed = self.fsm.parse(command)

  953             from lightweight_vla_adapter.src.driving_plan_runtime import DrivingPlanRuntime
  954             if self.driving_plan is None:self.driving_plan=DrivingPlanRuntime()

  956             observed_execution=feedback_reader() if callable(feedback_reader) else {}

  958                 ego_heading_deg=float(self.ego.get_transform().rotation.yaw))

  960                 timestamp_s=float(self.world.get_snapshot().timestamp.elapsed_seconds),
  961                 speed_mps=_speed_mps(self.ego),execution_state=observed_execution)
  962             step=step or plan_document['intent']['steps'][-1]
  963             plan_step_id=step['step_id']
  964             parsed=self.fsm.parsed_step(step,parsed.source_text)
  965             command=dict(command,id=plan_document['request_id']+':'+plan_step_id)
  966         else:self.driving_plan=None
  967         parsed=self.fsm.resolve_relative_speed(
  968             command,
  969             parsed,
  970             ego_speed_kmh=3.6*_speed_mps(self.ego),
  971         )
  972         set_cruise=getattr(self.route_controller,'set_cruise_target_kmh',None)
  973         if callable(set_cruise):
  974             set_cruise(parsed.target_speed_kmh if parsed.parsed_intent in
  975                        ('SET_SPEED','DECELERATE','RESUME') else None)
  976         parsed,lane_corridor_points=self._lane_command_observation(command,parsed)
  977         intent_key = parsed.parsed_intent
  978         if parsed.requested_lane_direction is not None:
  979             intent_key = f"{parsed.parsed_intent}_{parsed.requested_lane_direction}"
  980         if plan_step_id is not None:intent_key+=':'+plan_step_id
  981         if intent_key != self._last_intent:
  982             self.pipeline.reset_temporal_state()
  983             self.supervisor.reset()
  984             self.lane_risk_contract.reset()
  985             self._last_intent = intent_key
  986
  987         tokens, text_mask = self.fsm.encode_tokens(
  988             parsed,
  989             cache_key=f"{intent_key}:{parsed.target_speed_kmh}",
  990         )
  991         frame_id = f"carla_{frame}"
  992         policy_state = build_sensor_policy_state(
  993             self.world, self.ego, frame_id=frame_id
  994         )
  995         timestamp_s = float(policy_state.get("timestamp_s", 0.0))
  996
  997         if self.enable_lidar:

 1577     def run_step(self) -> Any:
 1578         self._resource_journal.start()
 1579         if self._execution_journal is not None:
 1580             self._execution_journal.observe(self.world, self.ego)
 1581         try:
 1582             frame = int(self.world.get_snapshot().frame)
 1583             if frame - self._last_frame >= self.decision_interval_frames:
 1584                 self._decide(frame)
 1585                 self._last_frame = frame
 1586         except (OSError, RuntimeError, TypeError, ValueError, KeyError) as error:
 1587             if is_initial_sensor_warmup_error(error, self._decision_count):
 1588                 self._sensor_warmup_hold_count += 1
 1589                 self.route_controller.set_high_level_decision(
 1590                     {
 1591                         "action": "stop",
 1592                         "target_speed_kmh": 0.0,
 1593                         "emergency": False,
 1594                         "reason": "vla_sensor_warmup_safe_hold",
 1595                     }
 1596                 )
 1597                 self._last_overlay = {
 1598                     "asr_text": "Waiting for synchronized VLA sensors",
 1599                     "action": "stop",
 1600                     "target_speed_kmh": 0.0,
 1601                     "emergency": False,
 1602                     "risk_level": "UNKNOWN",
 1603                     "policy_state": "SENSOR_WARMUP",
 1604                 }
 1605                 self._stream.write(
 1606                     json.dumps(
 1607                         {
 1608                             "status": "sensor_warmup_safe_hold",
 1609                             "error_type": type(error).__name__,
 1610                             "error": str(error),
 1611                         },
 1612                         ensure_ascii=False,
 1613                     )
 1614                     + "\n"
 1615                 )
 1616                 self._stream.flush()
 1617                 return self.route_controller.run_step()
 1618             self._fallback_count += 1
 1619             self.route_controller.set_high_level_decision(
 1620                 {
 1621                     "action": "stop",
