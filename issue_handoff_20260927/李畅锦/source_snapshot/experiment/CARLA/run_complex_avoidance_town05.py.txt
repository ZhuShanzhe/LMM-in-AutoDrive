  281 def build_vla_command_schedule(
  282     config: Mapping[str, Any],
  283 ) -> list[dict[str, Any]]:
  284     """Compile reviewed Scene-2 commands into compound DrivingIntents."""
  285
  286     result = []
  287     for command in config["commands"]:
  288         intent = build_scheduled_driving_intent(
  289             command,
  290             simulation_frame=0,
  291             route_s_m=float(command["announce_at_m"]),
  292             timestamp_s=0.0,
  293         )
  294         intent["request_id"] = "{0}-configured".format(command["id"])
  295         result.append(
  296             {
  297                 "id": str(command["id"]),
  298                 "announce_at_m": float(command["announce_at_m"]),
  299                 "activate_at_m": float(command["announce_at_m"]),
  300                 "voice_text": str(command["spoken_text"]),
  301                 "target_speed_kmh": float(
  302                     config["route"]["target_speed_kmh"]
  303                 ),
  304                 "action": "keep_lane",
  305                 "driving_intent": intent,
  306             }
  307         )
  308     return result

  311 def vla_route_context(
  312     route: list[tuple[Any, Any]],
  313     route_distances: list[float],

  315     progress_m: float,
  316     simulation_time_s: float,
  317     default_speed_kmh: float,
  318 ) -> dict[str, Any]:
  319     """Return the same short-horizon geometry contract as the unified runner."""
  320
  321     target_index = bisect.bisect_left(route_distances, progress_m + 10.0)

  323     target_transform = route[target_index][0].transform

  325     return {
  326         "progress_m": float(progress_m),

  328         "default_speed_kmh": float(default_speed_kmh),
  329         "route_target": {

  331             "y": float(target_transform.location.y),

  350 def load_config(path: Path) -> dict[str, Any]:
  351     payload = json.loads(path.read_text(encoding="utf-8"))
  352     if payload.get("schema_version") != "scene_2_town05_runtime/v1":
  353         raise ValueError("unsupported Town05 Scene 2 config schema")
  354     if payload.get("map") != "Town05_Opt":
  355         raise ValueError("Town05 Scene 2 must use Town05_Opt")
  356     route = payload["route"]
  357     if float(route["target_length_m"]) < 8000.0:
  358         raise ValueError("Scene 2 route must be at least 8 km")
  359     if float(route.get("minimum_curvature_degrees", 0.0)) < 180.0:
  360         raise ValueError("Scene 2 route must require visible curved driving")
  361     if len(payload.get("commands", [])) != 15:
  362         raise ValueError("Scene 2 requires exactly 15 demonstration commands")
  363     for command in payload["commands"]:
  364         required_command_fields = {
  365             "id",
  366             "announce_at_m",
  367             "text",
  368             "spoken_text",
  369             "category",
  370             "urgency",
  371             "steps",
  372         }
  373         missing = required_command_fields.difference(command)
  374         if missing:
  375             raise ValueError(
  376                 "structured command fields are missing for {0}: {1}".format(
  377                     command.get("id", "unknown"),
  378                     sorted(missing),
  379                 )
  380             )
  381         if len(command["steps"]) < 2:
  382             raise ValueError(
  383                 "competition commands must contain ordered compound steps"
  384             )
  385         if str(command["text"]) != str(command["spoken_text"]):
  386             raise ValueError(
  387                 "command display text and spoken text must match exactly: "
  388                 + str(command["id"])
  389             )
  390     kinds = {
  391         str(event["kind"])
  392         for event in payload.get("special_events", [])
  393     }
  394     required = {
  395         "slow_vehicle",
  396         "crossing_pedestrian",
  397         "bus_stop",
  398         "cyclist",
  399     }
  400     if not required.issubset(kinds):
  401         raise ValueError(
  402             "special events must include slow vehicle, pedestrian, "
  403             "bus stop, and cyclist"
  404         )
  405     for event in payload.get("special_events", []):
  406         if len(event.get("variants", [])) < 2:
  407             raise ValueError(
  408                 "every Town05 special event requires at least two variants: "
  409                 + str(event.get("id"))
  410             )
  411     validate_event_ground_truth_contracts(payload["special_events"])
  412     traffic = payload["traffic"]
  413     if bool(traffic["hybrid_physics"]):
  414         raise ValueError(
  415             "hybrid physics is disabled to prevent visible NPC teleporting"
  416         )
  417     if bool(traffic["respawn_dormant_vehicles"]):
  418         raise ValueError(
  419             "runtime respawn is disabled to prevent visible NPC pop-in"
  420         )
  421     weather = payload["weather"]
  422     if (
  423         weather.get("preset") != "cloudy-evening"
  424         or float(weather["cloudiness"]) < 70.0
  425         or float(weather["sun_altitude_angle"]) > 10.0
  426     ):
  427         raise ValueError(
  428             "Scene 2 competition weather must be cloudy evening/low light"
  429         )
  430     sensors = payload["sensors"]
  431     if not set(REQUIRED_SENSOR_NAMES).issubset(sensors["required"]):
  432         raise ValueError("four RGB views and LiDAR are required")
  433     interfaces = payload["interfaces"]
  434     if not interfaces.get("simulation_frame_is_sync_key"):
  435         raise ValueError("simulation_frame must be the synchronization key")
  436     if interfaces.get("allow_adjacent_frame_fill"):
  437         raise ValueError("adjacent-frame modality filling is forbidden")
  438     acceptance = payload["acceptance"]
  439     if float(acceptance["asr_accuracy_min"]) < 0.96:
  440         raise ValueError("ASR acceptance must be at least 96%")
  441     if float(acceptance["semantic_alignment_accuracy_min"]) < 0.985:
  442         raise ValueError("semantic alignment acceptance must be at least 98.5%")
  443     if int(acceptance["violation_count_at_most"]) > 1:
  444         raise ValueError("competition allows at most one violation")
  445     return payload

  448 def road_option_name(value: Any) -> str:
  449     # CARLA releases expose RoadOption as either Enum or IntEnum.  On modern
  450     # Python, str(IntEnumMember) is the numeric value (for example ``2``), so
  451     # parsing only its string representation silently loses every maneuver.
  452     enum_name = getattr(value, "name", None)
  453     if enum_name:
  454         return str(enum_name).upper()
  455     return str(value).rsplit(".", 1)[-1].upper()

  458 def route_aware_preview_speed_kmh(
  459     progress_m: float,
  460     maneuvers: list[Mapping[str, Any]],
  461     cruise_speed_kmh: float,
  462     turn_speed_kmh: float,
  463     approach_m: float,
  464     exit_m: float,
  465 ) -> float:
  466     """Cap preview speed around planned turns to avoid corner cutting."""
  467
  468     cruise = max(0.0, float(cruise_speed_kmh))
  469     turn_speed = min(cruise, max(0.0, float(turn_speed_kmh)))
  470     approach = max(0.0, float(approach_m))
  471     exit_distance = max(0.0, float(exit_m))
  472     progress = float(progress_m)
  473     for maneuver in maneuvers:
  474         if str(maneuver.get("route_option", "")).upper() not in {
  475             "LEFT",
  476             "RIGHT",
  477         }:
  478             continue
  479         delta_m = float(maneuver["progress_m"]) - progress
  480         if -exit_distance <= delta_m <= approach:
  481             return turn_speed
  482     return cruise

  485 def planned_turn_window_active(
  486     progress_m: float,
  487     maneuvers: list[Mapping[str, Any]],
  488     approach_m: float,
  489     exit_m: float,
  490 ) -> bool:
  491     """Return whether progress is inside a planned turn control window."""
  492
  493     progress = float(progress_m)
  494     approach = max(0.0, float(approach_m))
  495     exit_distance = max(0.0, float(exit_m))
  496     return any(
  497         str(maneuver.get("route_option", "")).upper()
  498         in {"LEFT", "RIGHT"}
  499         and -exit_distance
  500         <= float(maneuver["progress_m"]) - progress
  501         <= approach
  502         for maneuver in maneuvers
  503     )

  506 def route_centering_steer_correction(
  507     ego_location: Any,
  508     route_waypoint: Any,
  509     lateral_gain: float,
  510     maximum_correction: float,
  511 ) -> tuple[float, float]:
  512     """Return a bounded steering correction toward the route centerline.
  513
  514     CARLA steering is positive to the right.  Signed lateral error is also
  515     positive to the waypoint's right, so the stabilizing correction has the
  516     opposite sign.  This guard is applied only around audited route turns.
  517     """
  518
  519     transform = route_waypoint.transform
  520     center = transform.location
  521     yaw = math.radians(float(transform.rotation.yaw))
  522     right_x = -math.sin(yaw)
  523     right_y = math.cos(yaw)
  524     dx = float(ego_location.x) - float(center.x)
  525     dy = float(ego_location.y) - float(center.y)
  526     lateral_error_m = dx * right_x + dy * right_y
  527     limit = max(0.0, float(maximum_correction))
  528     correction = -float(lateral_gain) * lateral_error_m
  529     correction = max(-limit, min(limit, correction))
  530     return correction, lateral_error_m

  533 def audit_command_route_alignment(
  534     commands: list[Mapping[str, Any]],
  535     route: list[tuple[Any, Any]],
  536     route_distances: list[float],
  537     default_horizon_m: float = 350.0,
  538 ) -> dict[str, Any]:
  539     """Check explicit upcoming turn/straight steps against the GRP route."""
  540
  541     required_options = {
  542         "TURN:LEFT": "LEFT",
  543         "TURN:RIGHT": "RIGHT",
  544         "PROCEED:STRAIGHT_THROUGH_JUNCTION": "STRAIGHT",
  545     }
  546     maneuver_names = {"LEFT", "RIGHT", "STRAIGHT"}
  547     global_maneuvers = []
  548     previous_name = None
  549     for index, distance in enumerate(route_distances):
  550         name = road_option_name(route[index][1])
  551         if name in maneuver_names and name != previous_name:
  552             global_maneuvers.append(
  553                 {
  554                     "route_option": name,
  555                     "progress_m": round(float(distance), 3),
  556                 }
  557             )
  558         previous_name = name
  559
  560     records = []
  561     for command in commands:
  562         required = [
  563             required_options[step]
  564             for step in command.get("steps", [])
  565             if step in required_options
  566         ]
  567         start_m = float(command["announce_at_m"])
  568         end_m = start_m + float(
  569             command.get("route_alignment_horizon_m", default_horizon_m)
  570         )
  571         maneuvers = [
  572             maneuver
  573             for maneuver in global_maneuvers
  574             if start_m <= maneuver["progress_m"] <= end_m
  575         ]
  576         encountered = [
  577             maneuver["route_option"] for maneuver in maneuvers
  578         ]
  579         cursor = iter(encountered)
  580         matched = all(
  581             any(actual == expected for actual in cursor)
  582             for expected in required
  583         )
  584         records.append(
  585             {
  586                 "command_id": str(command["id"]),
  587                 "announce_at_m": start_m,
  588                 "horizon_end_m": end_m,
  589                 "required_route_options": required,
  590                 "encountered_route_options": encountered,
  591                 "maneuvers": maneuvers,
  592                 "matched": matched,
  593             }
  594         )
  595     mismatches = [
  596         record for record in records if not record["matched"]
  597     ]
  598     return {
  599         "schema_version": "scene_2_route_command_audit/v1",
  600         "commands_audited": len(records),
  601         "mismatch_count": len(mismatches),
  602         "competition_ready": not mismatches,
  603         "global_maneuvers": global_maneuvers,
  604         "records": records,
  605     }

  663     def __post_init__(self) -> None:
  664         # ``sensor.other.collision`` emits one callback for every contact
  665         # frame.  Keep those raw samples for diagnostics, but group adjacent
  666         # samples from the same actor into one physical collision episode.
  667         self.collision_samples: list[dict[str, Any]] = []
  668         self.collisions: list[dict[str, Any]] = []
  669         self._open_collision_by_actor: dict[str, dict[str, Any]] = {}
  670         self.lane_invasions: list[dict[str, Any]] = []
  671         self.simulation_time_s = 0.0
  672
  673     def start(self) -> None:
  674         import carla
  675
  676         library = self.world.get_blueprint_library()
  677         collision = self.world.spawn_actor(
  678             library.find("sensor.other.collision"),
  679             carla.Transform(),
  680             attach_to=self.ego,
  681         )
  682         lane = self.world.spawn_actor(
  683             library.find("sensor.other.lane_invasion"),
  684             carla.Transform(),
  685             attach_to=self.ego,
  686         )
  687         collision.listen(self._on_collision)
  688         lane.listen(self._on_lane_invasion)
  689         self.registry.add(collision)
  690         self.registry.add(lane)

  692     def _on_collision(self, event: Any) -> None:
  693         other = getattr(event, "other_actor", None)
  694         frame = int(event.frame)
  695         actor_id = getattr(other, "id", None)
  696         actor_type = getattr(other, "type_id", "unknown")
  697         actor_attributes = getattr(other, "attributes", {}) or {}
  698         actor_role = actor_attributes.get("role_name")
  699         sample = {
  700             "frame": frame,
  701             "simulation_time_s": self.simulation_time_s,
  702             "other_actor_id": actor_id,
  703             "other_actor_type": actor_type,
  704             "other_actor_role": actor_role,
  705         }
  706         self.collision_samples.append(sample)
  707
  708         actor_key = (
  709             "id:{0}".format(actor_id)
  710             if actor_id is not None
  711             else "type:{0}".format(actor_type)
  712         )
  713         episode = self._open_collision_by_actor.get(actor_key)
  714         if (
  715             episode is not None
  716             and frame - int(episode["last_frame"]) <= 10
  717         ):
  718             episode["last_frame"] = frame
  719             episode["last_time_s"] = self.simulation_time_s
  720             episode["contact_samples"] += 1
  721             return
  722
  723         episode = {
  724             "first_frame": frame,
  725             "last_frame": frame,
  726             "first_time_s": self.simulation_time_s,
  727             "last_time_s": self.simulation_time_s,
  728             "other_actor_id": actor_id,
  729             "other_actor_type": actor_type,
  730             "other_actor_role": actor_role,
  731             "contact_samples": 1,
  732         }
  733         self.collisions.append(episode)
  734         self._open_collision_by_actor[actor_key] = episode

  865 def command_event_requirements_met(
  866     command: Mapping[str, Any],
  867     event_states: Mapping[str, str],
  868 ) -> bool:
  869     """Return whether all event-state gates for a command are satisfied."""
  870
  871     requirements = command.get("requires_event_states", {})
  872     if requirements is None:
  873         return True
  874     if not isinstance(requirements, Mapping):
  875         raise ValueError(
  876             "command requires_event_states must be a mapping"
  877         )
  878     return all(
  879         event_states.get(str(event_id)) == str(required_state)
  880         for event_id, required_state in requirements.items()
  881     )

  884 def commands_for_segment(commands, start_progress_m):
  885     """Cold segment runs omit earlier commands, without claiming they ran."""
  886     start=float(start_progress_m)
  887     if not math.isfinite(start) or start<0:
  888         raise ValueError('segment start must be finite and nonnegative')
  889     selected,skipped=[],[]
  890     previous=-1.0
  891     identities=set()
  892     for command in commands:
  893         position=float(command['announce_at_m'])
  894         identity=str(command['id'])
  895         if not math.isfinite(position) or position<0 or position<previous or identity in identities:
  896             raise ValueError('command schedule must be finite, ordered and uniquely identified')
  897         previous=position
  898         identities.add(identity)
  899         if position<start:
  900             skipped.append(identity)
  901         else:
  902             selected.append(command)
  903     return selected,skipped

  906 def ready_commands_in_order(
  907     commands: list[Mapping[str, Any]],
  908     announced: set[str],
  909     progress_m: float,
  910     event_states: Mapping[str, str],
  911 ) -> list[Mapping[str, Any]]:
  912     """Return the contiguous, ordered prefix ready for announcement.
  913
  914     A command whose progress or event gate is not ready blocks every later
  915     command.  This preserves the competition requirement that compound
  916     instructions are delivered and evaluated in their configured order.
  917     """
  918
  919     ready = []
  920     for command in commands:
  921         command_id = str(command["id"])
  922         if command_id in announced:
  923             continue
  924         if progress_m < float(command["announce_at_m"]):
  925             break
  926         if not command_event_requirements_met(command, event_states):
  927             break
  928         ready.append(command)
  929     return ready
