    1 """Completion-gated command dispatch without simulator assessment truth."""
    2 from copy import deepcopy
    3 import math
    4
    5
    6 def resolve_command_dispatch_mode(commands, configured_mode=None):
    7     """Choose completion gating only when every command carries a usable plan."""
    8
    9     if configured_mode is not None:
   10         if configured_mode not in ('route_latest', 'completion_serial'):
   11             raise ValueError('unknown command dispatch mode')
   12         return configured_mode
   13     commands = list(commands)
   14     if commands and all(
   15         isinstance(command.get('driving_intent'), dict)
   16         and isinstance(command['driving_intent'].get('request_id'), str)
   17         and bool(command['driving_intent']['request_id'])
   18         and bool(command['driving_intent'].get('intent', {}).get('steps'))
   19         for command in commands
   20     ):
   21         return 'completion_serial'
   22     return 'route_latest'
   23
   24
   25 class CompletionCommandQueue:
   26     def __init__(self, commands, timeout_s=120, safety_wait_timeout_s=300):
   27         if not math.isfinite(timeout_s) or timeout_s <= 0:
   28             raise ValueError('positive command timeout required')
   29         self.timeout_s=timeout_s
   30         if not math.isfinite(safety_wait_timeout_s) or safety_wait_timeout_s <= 0:
   31             raise ValueError('positive safety wait timeout required')
   32         self.safety_wait_timeout_s=safety_wait_timeout_s
   33         self.execution_elapsed_s=0.
   34         self.safety_wait_elapsed_s=0.
   35         self.commands=deepcopy(list(commands))
   36         seen=set()
   37         previous=-math.inf
   38         for command in self.commands:
   39             document=command.get('driving_intent') or {}
   40             identity=document.get('request_id')
   41             trigger=next((command[k] for k in ('trigger_progress_m','activate_at_m','announce_at_m') if k in command),0)
   42             if (not isinstance(identity,str) or not identity or identity in seen or
   43                     not document.get('intent',{}).get('steps')):
   44                 raise ValueError('serial dispatch requires distinct structured instruction requests')
   45             if isinstance(trigger,bool) or not isinstance(trigger,(int,float)) or not math.isfinite(trigger) or trigger<previous:
   46                 raise ValueError('serial command triggers must be finite and ordered')
   47             command['_dispatch_trigger_m']=float(trigger)
   48             end=next((command[k] for k in ('end_progress_m','deactivate_at_m') if command.get(k) is not None),None)
   49             if end is not None and (isinstance(end,bool) or not isinstance(end,(int,float))
   50                                     or not math.isfinite(end) or end<=trigger):
   51                 raise ValueError('command end progress must be finite and greater than trigger')
   52             command['_dispatch_end_m']=end
   53             previous=trigger
   54             seen.add(identity)
   55         self.index=0
   56         self.current=None
   57         self.started=None
   58         self.last_time=None
   59         self.failed=None
   60         self.events=[]
   61
   62     def select(self, progress_m, timestamp_s, feedback=None):
   63         if not math.isfinite(progress_m) or not math.isfinite(timestamp_s):
   64             raise ValueError('finite dispatch progress/time required')
   65         if self.last_time is not None and timestamp_s < self.last_time:
   66             raise ValueError('dispatch time regressed')
   67         previous_time=self.last_time
   68         self.last_time=timestamp_s
   69         if self.failed:
   70             raise ValueError('command dispatch halted: '+self.failed)
   71         if self.current is not None:
   72             identity=self.current['driving_intent']['request_id']
   73             end=self.commands[self.index]['_dispatch_end_m']
   74             matching=bool(feedback and feedback.get('request_id')==identity)
   75             observed=feedback.get('observed_at_s') if matching else None
   76             fresh=(isinstance(observed,(int,float)) and not isinstance(observed,bool)
   77                    and math.isfinite(observed) and previous_time is not None
   78                    and previous_time-1e-6 <= observed <= timestamp_s
   79                    and timestamp_s-observed <= .5)
   80             elapsed=max(0.,timestamp_s-previous_time) if previous_time is not None else 0.
   81             if fresh and feedback.get('safety_wait') is True and feedback.get('plan_status')=='ACTIVE':
   82                 self.safety_wait_elapsed_s+=elapsed
   83             else:
   84                 self.execution_elapsed_s+=elapsed
   85             if end is not None and progress_m>=end:
   86                 self.failed='ACTIVE_COMMAND_WINDOW_EXPIRED'
   87             elif self.execution_elapsed_s>=self.timeout_s:
   88                 self.failed='TIMEOUT'
   89             elif self.safety_wait_elapsed_s>=self.safety_wait_timeout_s:
   90                 self.failed='SAFETY_WAIT_TIMEOUT'
   91             if not self.failed and feedback and feedback.get('request_id')==identity:
   92                 status=feedback.get('plan_status')
   93                 if status=='COMPLETED':
   94                     self.events.append(dict(request_id=identity,status='COMPLETED',timestamp_s=timestamp_s))
   95                     self.current=None
   96                     self.index+=1
   97                 elif status in ('FAILED','CANCELLED'):
   98                     self.failed=status
   99             if not self.failed and self.current is not None:
  100                 if self.safety_wait_elapsed_s>=self.safety_wait_timeout_s:
  101                     self.failed='SAFETY_WAIT_TIMEOUT'
  102                 elif self.execution_elapsed_s>=self.timeout_s:
  103                     self.failed='TIMEOUT'
  104             if self.failed:
  105                 self.events.append(dict(request_id=identity,status=self.failed,timestamp_s=timestamp_s,
  106                                         progress_m=progress_m,end_progress_m=end))
  107                 raise ValueError('command dispatch halted: '+self.failed)
  108         if self.current is None and self.index<len(self.commands):
  109             candidate=self.commands[self.index]
  110             if progress_m>=candidate['_dispatch_trigger_m']:
  111                 end=candidate['_dispatch_end_m']
  112                 if end is not None and progress_m>=end:
  113                     self.failed='MISSED_COMMAND_WINDOW'
  114                     self.events.append(dict(request_id=candidate['driving_intent']['request_id'],
  115                                             status=self.failed,timestamp_s=timestamp_s))
  116                     raise ValueError('command dispatch halted: '+self.failed)
  117                 self.current=deepcopy(candidate)
  118                 self.current.pop('_dispatch_trigger_m')
  119                 self.current.pop('_dispatch_end_m')
  120                 self.started=timestamp_s
  121                 self.execution_elapsed_s=0.
  122                 self.safety_wait_elapsed_s=0.
  123                 self.events.append(dict(request_id=self.current['driving_intent']['request_id'],
  124                                         status='RUNNING',timestamp_s=timestamp_s,progress_m=progress_m))
  125         return deepcopy(self.current) if self.current else {'text':'Continue driving safely in the current lane.'}
  126
  127     def status(self):
  128         return dict(mode='completion_serial',completed_commands=self.index,total_commands=len(self.commands),
  129                     active_request_id=self.current['driving_intent']['request_id'] if self.current else None,
  130                     failure=self.failed,events=deepcopy(self.events),
  131                     execution_elapsed_s=self.execution_elapsed_s,
  132                     safety_wait_elapsed_s=self.safety_wait_elapsed_s,
  133                     feedback_scope='model_execution_plan; not independent task assessment')
