    1 """Generic route-following PID used by every CARLA scene.
    2
    3 The PID consumes one high-level decision dict and converts it into a
    4 ``carla.VehicleControl``.  It owns no scene/event/command semantics: lane
    5 changes, target speed and hazard responses always come from the decision
    6 produced by the universal VLA chain.
    7 """
    8
    9 from __future__ import annotations
   10
   11 import bisect
   12 import math
   13 from typing import Any, Mapping, Sequence
   14
   15 import carla

   21 class GenericRoutePID:
   22     """Low-level route follower over a CARLA waypoint or point route."""
   23
   24     def __init__(
   25         self,
   26         world: Any,
   27         ego: Any,
   28         *,
   29         target_speed_kmh: float,
   30         fixed_delta_seconds: float,
   31         route_context: Any | None = None,
   32         route_plan: Sequence[tuple[Any, Any]] | None = None,
   33         route_manager: Any | None = None,
   34         logical_lane_adapter: Any | None = None,
   35     ) -> None:
   36         if route_context is not None and route_plan is None:
   37             raise ValueError("route_plan is required with route_context")
   38         self.world = world
   39         self.ego = ego
   40         self.route_context = route_context
   41         self.route_plan = list(route_plan) if route_plan is not None else None
   42         self.route_manager = route_manager
   43         self.logical_lane_adapter = logical_lane_adapter
   44         self.fixed_delta_seconds = float(fixed_delta_seconds)
   45         self.target_speed_kmh = float(target_speed_kmh)
   46         self._default_speed_kmh = float(target_speed_kmh)
   47         self._requested_cruise_speed_kmh = float(target_speed_kmh)
   48         self._action = "keep_lane"
   49         self._target_speed = float(target_speed_kmh)
   50         self._target_lane: str | None = None
   51         self._emergency = False
   52         self._sequence_command = {}
   53         self._sequence_expiry = None
   54         self._pid = EgoPIDController(
   55             ego,
   56             world.get_map(),
   57             float(target_speed_kmh),
   58         )

   60     def set_cruise_target_kmh(self, target: float | None) -> float:
   61         """Keep the requested cruise goal distinct from the current road limit."""
   62         if target is not None:
   63             if isinstance(target,bool) or not math.isfinite(float(target)) or not 0 <= float(target) <= 120:
   64                 raise ValueError('Cruise target must be a finite speed from 0 to 120 km/h')
   65             self._requested_cruise_speed_kmh=float(target)
   66         requested=getattr(self,'_requested_cruise_speed_kmh',self._default_speed_kmh)
   67         getter=getattr(getattr(self,'ego',None),'get_speed_limit',None)
   68         try:limit=float(getter()) if callable(getter) else float('nan')
   69         except (TypeError,ValueError,RuntimeError):limit=float('nan')
   70         if not math.isfinite(limit) or limit<=0:
   71             limit=self._default_speed_kmh
   72         self.target_speed_kmh=max(0.,min(requested,limit,120.))
   73         return self.target_speed_kmh
   74
   75     def set_high_level_decision(self, decision: Mapping[str, Any]) -> None:
   76         """Store one gated VLA decision for the current control step."""
   77
   78         self._action = str(decision.get("action", "keep_lane"))
   79         self._source_step_id=decision.get('source_step_id')
   80         self._command_id=decision.get('command_id') or self._action+':'+str(self._source_step_id)
   81         self._allow_positive_acceleration=bool(decision.get('allow_positive_acceleration',True))
   82         self._longitudinal_control_mode=decision.get('longitudinal_control_mode','ABSOLUTE_SPEED')
   83         try:
   84             speed = float(decision.get("target_speed_kmh", 0.0))
   85         except (TypeError, ValueError):
   86             speed = 0.0
   87         self._target_speed = max(
   88             0.0, min(self.set_cruise_target_kmh(None), speed)
   89         )
   90         if self._action in {"stop", "emergency_brake"}:
   91             self._target_speed = 0.0
   92         self._target_lane = decision.get("target_lane")
   93         self._emergency = bool(decision.get("emergency", False))
   94         self._sequence_command = {}
   95         self._sequence_expiry = None
   96         control_expiry = decision.get('control_valid_until_s')
   97         if control_expiry is not None:
   98             try:
   99                 control_expiry = float(control_expiry)
  100                 if not math.isfinite(control_expiry):raise ValueError('Nonfinite expiry')
  101                 self._sequence_expiry = control_expiry
  102             except (TypeError,ValueError):
  103                 self._action, self._target_speed = 'stop', 0.
  104         if decision.get('longitudinal_sequence_schema') == 'longitudinal_sequence/1.0':
  105             try:
  106                 acceleration = float(decision['target_acceleration_mps2'])
  107                 expiry = float(decision['sequence_valid_until_s'])
  108                 valid = math.isfinite(speed) and math.isfinite(acceleration) and -8.0001 <= acceleration <= 3.0001 and math.isfinite(expiry)
  109             except (KeyError,TypeError,ValueError):
  110                 valid = False
  111             if not valid:
  112                 self._action, self._target_speed = 'stop', 0.
  113             else:
  114                 # A speed cap may cancel feedforward, but cannot extend command validity.
  115                 self._sequence_expiry = min(expiry,self._sequence_expiry) if self._sequence_expiry is not None else expiry
  116                 acceleration = max(-8.,min(3.,acceleration))
  117                 if abs(self._target_speed-speed) < 1e-6 and not self._emergency:
  118                     self._sequence_command = dict(longitudinal_sequence_schema='longitudinal_sequence/1.0',
  119                         target_acceleration_mps2=acceleration,sequence_valid_until_s=expiry)
  120
  121     def progress_m(self) -> float:
  122         if self.route_context is not None:
  123             return float(
  124                 self.route_context.distances_m[
  125                     self.route_context.tracker.index
  126                 ]
  127             )
  128         if self.route_manager is not None:
  129             return float(self.route_manager.progress_m)
  130         return 0.0

  132     def radar_corridor_polyline(
  133         self,
  134         horizon_m: float = 85.0,
  135     ) -> list[tuple[float, float]]:
  136         """Return the planner-owned centreline ahead for sensor gating.
  137
  138         The result contains only route geometry and the ego pose.  It does not
  139         enumerate actors or scenario events, so all three scenes can use the
  140         same physical radar corridor filter.
  141         """
  142
  143         location = self.ego.get_location()
  144         points = [(float(location.x), float(location.y))]
  145         progress_m = self.progress_m()
  146         horizon = progress_m + max(5.0, float(horizon_m))
  147         if self.route_context is not None and self.route_plan is not None:
  148             start = bisect.bisect_left(
  149                 self.route_context.distances_m, progress_m
  150             )
  151             end = bisect.bisect_right(
  152                 self.route_context.distances_m, horizon
  153             )
  154             for waypoint, _road_option in self.route_plan[start:end]:
  155                 waypoint_location = waypoint.transform.location
  156                 point = (
  157                     float(waypoint_location.x),
  158                     float(waypoint_location.y),
  159                 )
  160                 if math.hypot(
  161                     point[0] - points[-1][0], point[1] - points[-1][1]
  162                 ) > 0.05:
  163                     points.append(point)
  164         elif self.route_manager is not None:
  165             route = getattr(self.route_manager, "route", None) or []
  166             for route_point in route:
  167                 try:
  168                     distance_m = float(route_point["distance_m"])
  169                     point = (
  170                         float(route_point["x"]),
  171                         float(route_point["y"]),
  172                     )
  173                 except (KeyError, TypeError, ValueError):
  174                     continue
  175                 if distance_m < progress_m:
  176                     continue
  177                 if distance_m > horizon:
  178                     break
  179                 if math.hypot(
  180                     point[0] - points[-1][0], point[1] - points[-1][1]
  181                 ) > 0.05:
  182                     points.append(point)
  183         if len(points) < 2:
  184             target = self._route_target()
  185             if target is not None:
  186                 point = (float(target["x"]), float(target["y"]))
  187                 if math.hypot(
  188                     point[0] - points[-1][0], point[1] - points[-1][1]
  189                 ) > 0.05:
  190                     points.append(point)
  191         return points

  193     def _update_progress(self) -> float:
  194         if self.route_context is not None:
  195             self.route_context.tracker.update(self.ego.get_location())
  196             return float(
  197                 self.route_context.distances_m[
  198                     self.route_context.tracker.index
  199                 ]
  200             )
  201         if self.route_manager is not None:
  202             return float(self.route_manager.update(self.ego.get_location()))
  203         return 0.0

  205     def _current_logical_lane(self, waypoint: Any) -> int | None:
  206         if self.logical_lane_adapter is None:
  207             return None
  208         progress_m = self.progress_m()
  209         for logical_lane in (-1, -2, -3):
  210             if self.logical_lane_adapter.waypoint_matches_logical_lane(
  211                 waypoint,
  212                 logical_lane,
  213                 progress_m,
  214             ):
  215                 return logical_lane
  216         return None
  217
  218     def _plan_logical_lane(self, progress_m: float) -> int | None:
  219         """Resolve the plan waypoint's logical lane at a route position."""
  220
  221         if self.logical_lane_adapter is None:
  222             return None
  223         try:
  224             planned = self.route_context.adapter.route_waypoint(progress_m)
  225         except Exception:
  226             return None
  227         for logical_lane in (-1, -2, -3):
  228             if self.logical_lane_adapter.waypoint_matches_logical_lane(
  229                 planned,
  230                 logical_lane,
  231                 progress_m,
  232             ):
  233                 return logical_lane
  234         return None

  236     def _target_logical_lane(self, progress_m: float) -> int | None:
  237         """Choose the corridor lane generically.
  238
  239         Follow the planned lane when it is stable over the next 30 m and
  240         differs from the current lane by at most one lane.  When the planned
  241         lane cannot be mapped to the logical lane set (for example a positive
  242         connector lane), return ``None`` so the caller follows the plan
  243         waypoint exactly.  This keeps the ego on the corridor without
  244         multi-lane plan jumps or event-derived lane windows.
  245         """
  246
  247         if self.logical_lane_adapter is None:
  248             return None
  249         try:
  250             current_waypoint = self.logical_lane_adapter.get_waypoint(
  251                 self.ego.get_location(),
  252                 project_to_road=True,
  253                 lane_type=carla.LaneType.Driving,
  254             )
  255         except RuntimeError:
  256             return None
  257         if current_waypoint is None:
  258             return None
  259         current_logical = self._current_logical_lane(current_waypoint)
  260         if current_logical is None:
  261             return None
  262         planned_lanes = {
  263             self._plan_logical_lane(progress_m + offset)
  264             for offset in (0.0, 15.0, 30.0)
  265         }
  266         if None in planned_lanes or len(planned_lanes) != 1:
  267             return None
  268         planned = next(iter(planned_lanes))
  269         if abs(planned - current_logical) <= 1:
  270             return planned
  271         return current_logical

  273     def _waypoint_target(self, lookahead_m: float) -> dict[str, float] | None:
  274         if self.route_context is None or self.route_plan is None:
  275             return None
  276         progress_m = self._update_progress()
  277         target_index = bisect.bisect_left(
  278             self.route_context.distances_m,
  279             progress_m + lookahead_m,
  280         )
  281         target_index = min(target_index, len(self.route_plan) - 1)
  282         planned = self.route_plan[target_index][0]
  283
  284         lane_change_active = self._action in {
  285             "lane_change_left",
  286             "lane_change_right",
  287         }
  288         if self.logical_lane_adapter is not None:
  289             current_waypoint = self.logical_lane_adapter.get_waypoint(
  290                 self.ego.get_location(),
  291                 project_to_road=True,
  292                 lane_type=carla.LaneType.Driving,
  293             )
  294             current_logical = (
  295                 self._current_logical_lane(current_waypoint)
  296                 if current_waypoint is not None
  297                 else None
  298             )
  299             target_logical = self._target_logical_lane(progress_m)
  300             if (
  301                 lane_change_active
  302                 and self._target_lane in {"left", "right"}
  303                 and current_logical is not None
  304             ):
  305                 # A commanded lane change is only executed when the planned
  306                 # corridor agrees with the commanded direction.  This keeps
  307                 # text-driven changes aligned with the route geometry (and
  308                 # avoids desynchronising the progress tracker by merging
  309                 # early into a lane the plan has not reached yet).
  310                 if self._target_lane == "left":
  311                     intended = (
  312                         current_logical
  313                         if current_logical == -1
  314                         else current_logical + 1
  315                     )
  316                 else:
  317                     intended = (
  318                         current_logical
  319                         if current_logical == -3
  320                         else current_logical - 1
  321                     )
  322                 plan_lane = self._plan_logical_lane(
  323                     progress_m + lookahead_m
  324                 )
  325                 if plan_lane == intended:
  326                     target_logical = intended
  327                 else:
  328                     target_logical = current_logical
  329             if target_logical is not None:
  330                 translated = self.logical_lane_adapter.logical_waypoint(
  331                     target_logical,
  332                     progress_m + lookahead_m,
  333                 )
  334                 if translated is not None:
  335                     planned = translated
  336         transform = planned.transform
  337         return {
  338             "x": float(transform.location.x),
  339             "y": float(transform.location.y),
  340             "z": float(transform.location.z),
  341             "yaw": float(transform.rotation.yaw),
  342         }
  343
  344     def _route_target(self) -> dict[str, Any] | None:
  345         speed_kmh = self._current_speed_kmh()
  346         # Speed-adaptive lookahead with a moderate floor.  Junction turns use
  347         # a shorter horizon derived from the current speed instead of a fixed
  348         # constant, so slow turns keep a tight arc while faster approaches
  349         # still look far enough ahead.
  350         lookahead_m = max(4.0, min(12.0, 3.5 + speed_kmh * 0.30))
  351         if self._junction_or_road_transition_ahead() and speed_kmh >= 3.0:
  352             # Short speed-adaptive horizon: tight junction turns need a close
  353             # pure-pursuit point, otherwise the approach cuts the inside edge.
  354             lookahead_m = min(
  355                 lookahead_m,
  356                 max(2.8, min(4.5, 2.2 + speed_kmh * 0.10)),
  357             )
  358         if self.route_context is not None:
  359             target = self._waypoint_target(lookahead_m)
  360             if target is None:
  361                 return None
  362             reference = self._waypoint_target(0.0) or target
  363             return {
  364                 **target,
  365                 "reference": reference,
  366             }
  367         if self.route_manager is not None:
  368             target = self.route_manager.target_point(lookahead_m)
  369             if target is None:
  370                 return None
  371             reference = self.route_manager.target_point(0.0) or target
  372             result = dict(target)
  373             result["reference"] = reference
  374             return result
  375         return None

  377     def _ahead_heading_change_deg(
  378         self, start_m: float = 8.0, span_m: float = 26.0
  379     ) -> float:
  380         """Absolute heading change of the planned corridor ahead (degrees)."""
  381
  382         try:
  383             if self.route_context is not None and self.route_plan is not None:
  384                 progress_m = self.progress_m()
  385
  386                 def _yaw(offset: float) -> float:
  387                     index = bisect.bisect_left(
  388                         self.route_context.distances_m,
  389                         progress_m + offset,
  390                     )
  391                     index = max(0, min(index, len(self.route_plan) - 1))
  392                     return math.radians(
  393                         float(
  394                             self.route_plan[index][0]
  395                             .transform.rotation.yaw
  396                         )
  397                     )
  398
  399                 yaw0 = _yaw(start_m)
  400                 yaw1 = _yaw(start_m + span_m)
  401             elif self.route_manager is not None:
  402                 point0 = self.route_manager.target_point(start_m)
  403                 point1 = self.route_manager.target_point(start_m + span_m)
  404                 if point0 is None or point1 is None:
  405                     return 0.0
  406                 yaw0 = math.radians(float(point0["yaw"]))
  407                 yaw1 = math.radians(float(point1["yaw"]))
  408             else:
  409                 return 0.0
  410         except (IndexError, KeyError, TypeError, ValueError, AttributeError):
  411             return 0.0
  412         return abs(math.degrees(self._pid._angle_delta(yaw1, yaw0)))

  414     def _current_speed_kmh(self) -> float:
  415         velocity = self.ego.get_velocity()
  416         return 3.6 * math.sqrt(
  417             float(velocity.x) ** 2
  418             + float(velocity.y) ** 2
  419             + float(velocity.z) ** 2
  420         )

  422     def _junction_or_road_transition_ahead(
  423         self,
  424         lookahead_m: float | None = None,
  425     ) -> bool:
  426         """Detect junctions or road-id transitions in the planned corridor."""
  427
  428         if lookahead_m is None:
  429             lookahead_m = _clamp(
  430                 20.0 + self._current_speed_kmh() * 0.40,
  431                 20.0,
  432                 50.0,
  433             )
  434         lookahead_m = float(lookahead_m)
  435         if self.route_context is None or self.route_plan is None:
  436             if self.route_manager is not None:
  437                 # Route-manager corridors expose the exact planned waypoints
  438                 # with junction flags; scanning them is faster and more
  439                 # reliable than walking the official map lanes.
  440                 route = getattr(self.route_manager, "route", None)
  441                 if route:
  442                     progress_m = float(
  443                         getattr(self.route_manager, "progress_m", 0.0) or 0.0
  444                     )
  445                     horizon = progress_m + float(lookahead_m)
  446                     for point in route:
  447                         distance_m = float(point.get("distance_m", -1.0))
  448                         if distance_m < progress_m - 2.0:
  449                             continue
  450                         if distance_m > horizon:
  451                             break
  452                         if bool(point.get("is_junction", False)):
  453                             return True
  454                 return False
  455             return False
  456         progress_m = self.progress_m()
  457         index = bisect.bisect_left(
  458             self.route_context.distances_m,
  459             progress_m,
  460         )
  461         end = bisect.bisect_left(
  462             self.route_context.distances_m,
  463             progress_m + float(lookahead_m),
  464         )
  465         end = min(end, len(self.route_plan) - 1)
  466         window = [
  467             waypoint
  468             for waypoint, _road_option in self.route_plan[index : end + 1]
  469         ]
  470         if any(
  471             bool(getattr(waypoint, "is_junction", False))
  472             for waypoint in window
  473         ):
  474             return True
  475         road_ids = {int(waypoint.road_id) for waypoint in window}
  476         return len(road_ids) > 1
  477
  478     def _lane_transition_ahead(self) -> bool:
  479         """True while the planned corridor moves to a different logical lane."""
  480
  481         if self.logical_lane_adapter is None or self.route_context is None:
  482             return False
  483         try:
  484             current_waypoint = self.logical_lane_adapter.get_waypoint(
  485                 self.ego.get_location(),
  486                 project_to_road=True,
  487                 lane_type=carla.LaneType.Driving,
  488             )
  489         except RuntimeError:
  490             return False
  491         if current_waypoint is None:
  492             return False
  493         current_logical = self._current_logical_lane(current_waypoint)
  494         if current_logical is None:
  495             return False
  496         planned = self._plan_logical_lane(self.progress_m() + 15.0)
  497         return planned is not None and planned != current_logical

  499     def autonomous_lane_change_legal(self, direction: str) -> bool:
  500         """Conservatively validate an uncommanded collision-escape lane.
  501
  502         Unlike instructed lane changes, uncertainty is fail-closed: the
  503         target lane must exist, be a driving lane and be reachable across a
  504         marking that explicitly permits the requested direction.
  505         """
  506
  507         if direction not in {"left", "right"}:
  508             raise ValueError("direction must be 'left' or 'right'")
  509         try:
  510             waypoint = self.world.get_map().get_waypoint(
  511                 self.ego.get_location(),
  512                 project_to_road=True,
  513                 lane_type=carla.LaneType.Driving,
  514             )
  515         except (RuntimeError, AttributeError, TypeError):
  516             return False
  517         if waypoint is None:
  518             return False
  519         try:
  520             permission = int(waypoint.lane_change)
  521             target = (
  522                 waypoint.get_left_lane()
  523                 if direction == "left"
  524                 else waypoint.get_right_lane()
  525             )
  526             if target is None or target.lane_type != carla.LaneType.Driving:
  527                 return False
  528             # Adjacent lanes travelling in the opposite direction are never
  529             # valid evasive targets even if malformed map metadata exposes
  530             # a lane-change bit.
  531             if int(target.lane_id) * int(waypoint.lane_id) <= 0:
  532                 return False
  533         except (RuntimeError, AttributeError, TypeError, ValueError):
  534             return False
  535         mask = 1 if direction == "left" else 2
  536         return bool(permission & mask)
  537
  538     def _lane_change_legal(self) -> bool:
  539         """Map-legality check for the requested lane change.
  540
  541         Uses only static map geometry (waypoint lane-change permissions), not
  542         actor truth.  A change across a solid marking is never executed; the
  543         intent remains active and proceeds when the corridor reaches a legal
  544         window (or the supervisor timeout keeps the ego crawling).
  545         """
  546
  547         if self._action not in ("lane_change_left", "lane_change_right"):
  548             return True
  549         direction = self._action.removeprefix("lane_change_")
  550         try:
  551             waypoint = self.world.get_map().get_waypoint(
  552                 self.ego.get_location(),
  553                 project_to_road=True,
  554                 lane_type=carla.LaneType.Driving,
  555             )
  556         except (RuntimeError, AttributeError, TypeError):
  557             return True
  558         if waypoint is None:
  559             return True
  560         try:
  561             permission = int(waypoint.lane_change)
  562         except (AttributeError, TypeError, ValueError):
  563             return True
  564         if direction == "left":
  565             return bool(permission & 1)
  566         return bool(permission & 2)

  568     def run_step(self) -> carla.VehicleControl:
  569         if not self._lane_change_legal():
  570             # Hold the lateral action in the current lane; the high-level
  571             # intent stays untouched so it can resume at a legal window.
  572             self._target_lane = None
  573             self._action = "keep_lane"
  574         intent = {
  575             "schema_version": "1.0.0",
  576             "action": self._action,
  577             "target_speed_kmh": self._target_speed,
  578             "target_lane": self._target_lane,
  579             "target_location": None,
  580             "emergency": self._emergency,
  581             "route_target_trusted": False,
  582             "allow_positive_acceleration": getattr(self,'_allow_positive_acceleration',True),
  583             "longitudinal_control_mode": getattr(self,'_longitudinal_control_mode','ABSOLUTE_SPEED'),
  584             "command_id": getattr(self,'_command_id',self._action),
  585         }
  586         if self._junction_or_road_transition_ahead():
  587             # Sharpness-aware junction ceiling: the cap falls with the actual
  588             # heading change of the planned corridor and rises slightly with
  589             # the approach speed.  It only limits physical speed and never
  590             # changes lane/turn selection.
  591             sharpness_deg = self._ahead_heading_change_deg()
  592             junction_cap = _clamp(
  593                 11.0 - 0.05 * sharpness_deg
  594                 + self._current_speed_kmh() * 0.05,
  595                 7.0,
  596                 14.0,
  597             )
  598             intent["target_speed_kmh"] = min(
  599                 float(intent["target_speed_kmh"]),
  600                 junction_cap,
  601             )
  602         if self._lane_transition_ahead():
  603             # Speed-adaptive lane-transition ceiling: faster approaches get a
  604             # lower cap so the lateral manoeuvre stays stable and cannot clip
  605             # lane boundaries; slow crawls keep enough momentum to finish the
  606             # change.
  607             transition_cap = _clamp(
  608                 16.0 - self._current_speed_kmh() * 0.10,
  609                 9.0,
  610                 16.0,
  611             )
  612             intent["target_speed_kmh"] = min(
  613                 float(intent["target_speed_kmh"]),
  614                 transition_cap,
  615             )
  616         route_target = self._route_target()
  617         if route_target is not None:
  618             intent["target_location"] = route_target
  619             intent["route_target_trusted"] = True
  620         # Route-manager corridors (e.g. scene 1) sample corners coarsely and
  621         # need additional steering authority while approaching or crossing a
  622         # junction; lane-centred route contexts keep the conservative
  623         # validated limit.
  624         intent["junction_steering_authority"] = bool(
  625             self.route_manager is not None
  626             and self._junction_or_road_transition_ahead()
  627         )
  628         expiry = self._sequence_expiry
  629         if expiry is not None and self.world.get_snapshot().timestamp.elapsed_seconds > expiry:
  630             intent.update(action='stop',target_speed_kmh=0.,emergency=False)
  631         elif self._sequence_command and abs(intent['target_speed_kmh']-self._target_speed)<1e-6:
  632             intent.update(self._sequence_command)
  633         control, _ = self._pid.run_step(intent, self.fixed_delta_seconds)
  634         return control
  635
  636     def execution_state(self) -> dict[str, Any]:
  637         return {
  638             "source_step_id": getattr(self,'_source_step_id',None),
  639             "action": self._action,
  640             "target_speed_kmh": self._target_speed,
  641             "target_lane": self._target_lane,
  642             "speed_kmh": round(self._current_speed_kmh(), 3),
  643             "progress_m": round(self.progress_m(), 3),
  644             "pid": self._pid.get_execution_state(),
  645         }
